lisääviholliset(id)
    viholliset = []
    vihollislista = readenemyjson(id)
    for vihollinen in vihollislista:
        switch(vihollinen.type)
            case 100:
                viholliset.append(new npcchar(vihollinen.x,vihollinen.y))
                return
            case 101:
                viholliset.append(new bosschar)
                return
            case 102:
                viholliset.append(new superbosschar)
                return
    return vihollislista
