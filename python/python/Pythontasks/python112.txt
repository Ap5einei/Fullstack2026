# osajono
osajono = "joki"
merkkijono = "Seinäjoki"

print("jo" in merkkkijono) # true

print(merkkijono.find("jo")) # ilmoittaa indeksin
print(merkkijono.find("xxx")) # ei löydy, palauttaa -1
